-- PointIQOS Auto 4.2 — record-level cloud sync (locked-down policies) -- Run ONCE in the Supabase SQL Editor (project lumkgbtopiztkylqaxit). -- -- Permission model (Option 1): -- * Owner AND salesperson can SEE and WORK every record. -- * Deletion is ALWAYS archived=true (a tombstone) — there is NO DELETE -- policy on any table, so neither role (nor any future code path) can -- hard-delete a row and bypass the tombstone. -- * Only the OWNER can set archived=true (enforced by trigger) and only -- the OWNER can write settings/portal URL (auto_kv). -- ---------- tables ---------- create table if not exists public.auto_contacts (id text primary key, data jsonb not null default '{}', assigned_to uuid, updated_at timestamptz not null default now(), archived boolean not null default false); create table if not exists public.auto_activity (id text primary key, data jsonb not null default '{}', contact_id text, updated_at timestamptz not null default now(), archived boolean not null default false); create table if not exists public.auto_credit_apps (id text primary key, data jsonb not null default '{}', contact_id text, updated_at timestamptz not null default now(), archived boolean not null default false); create table if not exists public.auto_sales (id text primary key, data jsonb not null default '{}', contact_id text, updated_at timestamptz not null default now(), archived boolean not null default false); create table if not exists public.auto_partners (id text primary key, data jsonb not null default '{}', updated_at timestamptz not null default now(), archived boolean not null default false); create table if not exists public.auto_kv (id text primary key, data jsonb not null default '{}', updated_at timestamptz not null default now()); -- ---------- owner check ---------- create or replace function public.auto_is_owner() returns boolean language sql stable security definer set search_path = public as $$ select exists (select 1 from public.profiles p where p.id = auth.uid() and lower(coalesce(p.role,'')) = 'owner'); $$; -- ---------- archive guard: only owner may set archived=true (insert or update) ---------- create or replace function public.auto_archive_guard() returns trigger language plpgsql as $$ begin -- Creating an archived/tombstone row requires Owner. if TG_OP = 'INSERT' then if coalesce(NEW.archived, false) and not public.auto_is_owner() then raise exception 'Only the owner can archive records'; end if; return NEW; end if; -- A tombstone is permanent. -- No user or stale device can change archived=true back to false. if coalesce(OLD.archived, false) and not coalesce(NEW.archived, false) then raise exception 'Archived records cannot be restored'; end if; -- Only Owner can change an active record into an archived record. if not coalesce(OLD.archived, false) and coalesce(NEW.archived, false) and not public.auto_is_owner() then raise exception 'Only the owner can archive records'; end if; return NEW; end; $$; -- ---------- RLS + policies + triggers for the 5 record tables ---------- -- NOTE: no DELETE policy is created. With RLS enabled and no DELETE policy, -- hard DELETE is denied for everyone. Deletion happens only via archived=true. do $$ declare tname text; begin foreach tname in array array['auto_contacts','auto_activity','auto_credit_apps','auto_sales','auto_partners'] loop execute format('alter table public.%I enable row level security;', tname); execute format('drop policy if exists %I on public.%I;', tname||'_sel', tname); execute format('drop policy if exists %I on public.%I;', tname||'_ins', tname); execute format('drop policy if exists %I on public.%I;', tname||'_upd', tname); execute format('drop policy if exists %I on public.%I;', tname||'_del', tname); -- remove any prior DELETE policy execute format('create policy %I on public.%I for select to authenticated using (true);', tname||'_sel', tname); execute format('create policy %I on public.%I for insert to authenticated with check (true);', tname||'_ins', tname); execute format('create policy %I on public.%I for update to authenticated using (true) with check (true);', tname||'_upd', tname); -- (intentionally NO delete policy) execute format('drop trigger if exists %I on public.%I;', tname||'_archguard', tname); execute format('create trigger %I before insert or update on public.%I for each row execute function public.auto_archive_guard();', tname||'_archguard', tname); end loop; end $$; -- ---------- auto_kv (settings / portal url): owner-managed ---------- alter table public.auto_kv enable row level security; drop policy if exists auto_kv_sel on public.auto_kv; drop policy if exists auto_kv_ins on public.auto_kv; drop policy if exists auto_kv_upd on public.auto_kv; create policy auto_kv_sel on public.auto_kv for select to authenticated using (true); create policy auto_kv_ins on public.auto_kv for insert to authenticated with check (public.auto_is_owner()); create policy auto_kv_upd on public.auto_kv for update to authenticated using (public.auto_is_owner()) with check (public.auto_is_owner()); -- ---------- 4.2: atomic archive RPC (contact + children in ONE transaction) ---------- create or replace function public.auto_archive_contact_bundle( p_contact_id text, p_activity_ids text[] default '{}'::text[], p_sale_ids text[] default '{}'::text[], p_credit_app_ids text[] default '{}'::text[] ) returns void language plpgsql security invoker set search_path = public as $$ declare v_ts timestamptz := now(); begin -- Explicit role check in addition to the archive trigger. if not public.auto_is_owner() then raise exception 'Only the owner can archive records'; end if; -- Contact tombstone. insert into public.auto_contacts (id, data, updated_at, archived) values (p_contact_id, jsonb_build_object('id', p_contact_id), v_ts, true) on conflict (id) do update set data = excluded.data, updated_at = excluded.updated_at, archived = true; -- Activity tombstones. if coalesce(array_length(p_activity_ids, 1), 0) > 0 then insert into public.auto_activity (id, data, updated_at, archived) select x.id, jsonb_build_object('id', x.id), v_ts, true from unnest(p_activity_ids) as x(id) on conflict (id) do update set data = excluded.data, updated_at = excluded.updated_at, archived = true; end if; -- Sale tombstones. if coalesce(array_length(p_sale_ids, 1), 0) > 0 then insert into public.auto_sales (id, data, updated_at, archived) select x.id, jsonb_build_object('id', x.id), v_ts, true from unnest(p_sale_ids) as x(id) on conflict (id) do update set data = excluded.data, updated_at = excluded.updated_at, archived = true; end if; -- Credit application tombstones. if coalesce(array_length(p_credit_app_ids, 1), 0) > 0 then insert into public.auto_credit_apps (id, data, updated_at, archived) select x.id, jsonb_build_object('id', x.id), v_ts, true from unnest(p_credit_app_ids) as x(id) on conflict (id) do update set data = excluded.data, updated_at = excluded.updated_at, archived = true; end if; end; $$; revoke all on function public.auto_archive_contact_bundle(text, text[], text[], text[]) from public; grant execute on function public.auto_archive_contact_bundle(text, text[], text[], text[]) to authenticated;
